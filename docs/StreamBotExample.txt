using System;
using System.IO;
using System.Net.Http;
using System.Text;
using System.Text.RegularExpressions;
using System.Threading.Tasks;

public class CPHInline
{
    private static readonly HttpClient client = new HttpClient();
    private static readonly HttpClient probeClient = new HttpClient { Timeout = TimeSpan.FromSeconds(1.5) };
    private static readonly string OutputDirectory = Path.Combine(Path.GetTempPath(), "TTS_Stream_Buffer");

    // Порядок = приоритет. Локальный первым.
    private static readonly string[] ServerUrls = { "http://127.0.0.1:8765", "http://192.168.2.60:8765" };
    private const int CacheSeconds = 30;
    private const int MaxLength = 300;
    private static int fileIndex = 0;

    public bool Execute()
    {
        try
        {
            string userName = "Зритель";
            if (CPH.TryGetArg("user", out string userArg) && !string.IsNullOrWhiteSpace(userArg)) userName = userArg;
            else if (CPH.TryGetArg("userName", out string uNameArg) && !string.IsNullOrWhiteSpace(uNameArg)) userName = uNameArg;

            string text;
            if (!CPH.TryGetArg("rawInput", out text) || string.IsNullOrWhiteSpace(text))
            {
                if (!CPH.TryGetArg("prompt", out text) || string.IsNullOrWhiteSpace(text))
                {
                    if (!CPH.TryGetArg("message", out text) || string.IsNullOrWhiteSpace(text))
                    {
                        CPH.SendMessage($"@{userName}, текст сообщения для озвучки не может быть пустым!");
                        return false;
                    }
                }
            }

            text = text.Trim();
            if (text.Length > MaxLength)
            {
                CPH.SendMessage($"@{userName}, твое сообщение слишком длинное ({text.Length} симв.)! Лимит — {MaxLength}.");
                return false;
            }

            string cleanText = CleanForTTS(text);
            if (string.IsNullOrWhiteSpace(cleanText)) return false;

            string json = "{\"text\":\"" + EscapeJson(cleanText) + "\",\"speed\":1.0}";

            byte[] wav = RequestWav(json);
            if (wav == null)
            {
                CPH.SendMessage($"@{userName}, TTS-сервер сейчас недоступен.");
                return false;
            }

            Directory.CreateDirectory(OutputDirectory);
            fileIndex = (fileIndex % 3) + 1;
            string outputFile = Path.Combine(OutputDirectory, "tts_" + fileIndex + ".wav");
            File.WriteAllBytes(outputFile, wav);

            CPH.PlaySound(outputFile, 1.0f, true, "MitaTTS", true);
            return true;
        }
        catch (Exception ex)
        {
            CPH.LogError("MITA TTS EXCEPTION: " + ex.ToString());
            return false;
        }
    }

    // Пробует закэшированный сервер, при сбое переопределяет и повторяет один раз
    private byte[] RequestWav(string json)
    {
        for (int attempt = 0; attempt < 2; attempt++)
        {
            string baseUrl = ResolveServer(attempt > 0);
            if (baseUrl == null) return null;
            try
            {
                var content = new StringContent(json, Encoding.UTF8, "application/json");
                var response = client.PostAsync(baseUrl + "/speak", content).GetAwaiter().GetResult();
                if (!response.IsSuccessStatusCode)
                {
                    CPH.LogError("TTS ответил кодом " + (int)response.StatusCode + " (" + baseUrl + ")");
                    return null; // сервер жив, но отказал: не перебираем
                }
                byte[] wav = response.Content.ReadAsByteArrayAsync().GetAwaiter().GetResult();
                return wav.Length < 1000 ? null : wav;
            }
            catch (HttpRequestException ex) { CPH.LogWarn("TTS недоступен (" + baseUrl + "): " + ex.Message); }
            catch (TaskCanceledException ex) { CPH.LogWarn("TTS таймаут (" + baseUrl + "): " + ex.Message); }
        }
        return null;
    }

    private string ResolveServer(bool force)
    {
        if (!force)
        {
            string cached = CPH.GetGlobalVar<string>("tts_base_url", false);
            long ts = CPH.GetGlobalVar<long>("tts_base_url_ts", false);
            if (!string.IsNullOrEmpty(cached) && (DateTime.UtcNow.Ticks - ts) < TimeSpan.FromSeconds(CacheSeconds).Ticks)
                return cached;
        }

        foreach (string baseUrl in ServerUrls)
        {
            try
            {
                var r = probeClient.GetAsync(baseUrl + "/api/system_config").GetAwaiter().GetResult();
                if (r.IsSuccessStatusCode)
                {
                    CPH.SetGlobalVar("tts_base_url", baseUrl, false);
                    CPH.SetGlobalVar("tts_base_url_ts", DateTime.UtcNow.Ticks, false);
                    return baseUrl;
                }
            }
            catch { }
        }
        return null;
    }

    private static string EscapeJson(string s) => s.Replace("\\", "\\\\").Replace("\"", "\\\"").Replace("\r", "\\r").Replace("\n", "\\n");

    private static string CleanForTTS(string s)
    {
        s = Regex.Replace(s, @"\bBuXaLo\b", "Бухало", RegexOptions.IgnoreCase);
        s = Regex.Replace(s, @"\p{Cs}|\p{So}|\p{Sk}|\p{Sm}|[\u200D\uFE0F\u20E3]", "");
        s = Regex.Replace(s, @"[\*""`_#\[\]\(\)<>~]", "");
        return Regex.Replace(s, @"\s+", " ").Trim();
    }
}
